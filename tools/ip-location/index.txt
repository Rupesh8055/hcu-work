function render_main() {
    $query = isset($_GET['ip']) ? trim($_GET['ip']) : '';
    echo '<div class="container">';
    echo '<h1>IP Location Finder</h1>';
    echo '<div class="desc">This tool will display geographic information about a supplied IP address including city, country, latitude, longitude and more.</div>';
    echo '<form class="search-box" onsubmit="fetchLocation(event)">';
    echo '<input type="text" id="ipInput" name="ip" placeholder="Enter IP address" value="' . htmlspecialchars($query) . '" required />';
    echo '<button type="submit">Find</button>';
    echo '</form>';
    echo '<div id="error" class="error" style="display: none;"></div>';
    echo '<div id="results" style="display: none;">';
    echo '<div class="results-header" id="resultsHeader"></div>';
    echo '<table class="results-table" id="resultsTable">';
    echo '<thead>';
    echo '<tr>';
    echo '<th>IP Address</th>';
    echo '<th>Country</th>';
    echo '<th>Region</th>';
    echo '<th>City</th>';
    echo '<th>Postal Code</th>';
    echo '<th>Latitude</th>';
    echo '<th>Longitude</th>';
    echo '<th>Timezone</th>';
    echo '</tr>';
    echo '</thead>';
    echo '<tbody id="resultsBody"></tbody>';
    echo '</table>';
    echo '</div>';
    echo '</div>';
}

function render_footer() {
    echo '<script>';
    echo 'async function fetchLocation(event) {';
    echo '    event.preventDefault();';
    echo '    const ip = document.getElementById("ipInput").value;';
    echo '    const url = ip ? `ip_location.php?ip=${encodeURIComponent(ip)}` : "ip_location.php";';
    echo '    const errorDiv = document.getElementById("error");';
    echo '    const resultsDiv = document.getElementById("results");';
    echo '    const resultsHeader = document.getElementById("resultsHeader");';
    echo '    const resultsBody = document.getElementById("resultsBody");';
    echo '    errorDiv.style.display = "none";';
    echo '    resultsDiv.style.display = "none";';
    echo '    try {';
    echo '        const response = await fetch(url);';
    echo '        const text = await response.text();'; // Log raw response
    echo '        console.log("Raw response:", text);'; // Debug raw response
    echo '        if (!response.ok) throw new Error(`HTTP error! Status: ${response.status}`);';
    echo '        const data = JSON.parse(text);'; // Parse JSON explicitly
    echo '        if (data.error) {';
    echo '            errorDiv.textContent = data.error;';
    echo '            errorDiv.style.display = "block";';
    echo '            return;';
    echo '        }';
    echo '        resultsHeader.textContent = `IP Location Results for ${ip}`;';
    echo '        resultsBody.innerHTML = `';
    echo '            <tr>';
    echo '                <td>${data.ip}</td>';
    echo '                <td>${data.country}</td>';
    echo '                <td>${data.region}</td>';
    echo '                <td>${data.city}</td>';
    echo '                <td>${data.postal_code}</td>';
    echo '                <td>${data.latitude}</td>';
    echo '                <td>${data.longitude}</td>';
    echo '                <td>${data.timezone}</td>';
    echo '            </tr>`;';
    echo '        resultsDiv.style.display = "block";';
    echo '    } catch (error) {';
    echo '        errorDiv.textContent = "Error fetching location data: " + error.message;';
    echo '        errorDiv.style.display = "block";';
    echo '    }';
    echo '}';
    echo '</script>';
    echo '</body></html>';
}

render_header();
render_main();
render_footer();
?>
